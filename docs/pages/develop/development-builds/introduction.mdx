class AnalyticsAndCRMService {
  private attioApiKey = 'ATTIO_API_KEY_LIVE';
  private baseUrl = 'https://api.attio.com/v2';

  /**
   * Syncs user gameplay traits & milestone conversions to Attio CRM
   */
  async syncUserToAttio(user: UserProfile, eventName: string, eventData?: Record<string, any>): Promise<void> {
    try {
      // 1. Assert user record in Attio
      await fetch(`${this.baseUrl}/objects/people/records`, {
        method: 'PUT',
        headers: {
          'Authorization': `Bearer ${this.attioApiKey}`,
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          data: {
            values: {
              email_addresses: [{ email_address: user.email }],
              name: [{ full_name: user.name }],
              custom_attributes: {
                streak_count: user.streakCount,
                is_pro_member: user.isPro,
                pearl_balance: user.pearlsBalance,
                last_event: eventName,
                last_event_meta: JSON.stringify(eventData || {}),
              },
            },
          },
        }),
      });
      console.log(`[Attio CRM] Synced event: ${eventName} for ${user.email}`);
    } catch (err) {
      console.error('[Attio CRM Error]', err);
    }
  }
}
class UGCCreatorService {
  private sideshiftApiKey = 'SK_LIVE_SIDESHIFT_API_KEY';
  private baseUrl = 'https://app.sideshift.app/api/embed';

  /**
   * Registers a UGC Creator inside SideShift for performance payouts
   */
  async registerUGCCreator(name: string, email: string, externalId: string): Promise<string | null> {
    try {
      const response = await fetch(`${this.baseUrl}/accounts/create`, {
        method: 'POST',
        headers: {
          'x-api-key': this.sideshiftApiKey,
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({ email, name, externalId }),
      });
      const data = await response.json();
      return data.sideshiftAccountId; // Save this ID to track attribution
    } catch (error) {
      console.error('[SideShift API Error]', error);
      return null;
    }
  }

  /**
   * Triggers automated revenue share payout to creator when a referred user subscribes
   */
  async processCreatorCommission(sideshiftAccountId: string, amountUsd: number): Promise<boolean> {
    try {
      const response = await fetch(`${this.baseUrl}/transfers/create`, {
        method: 'POST',
        headers: {
          'x-api-key': this.sideshiftApiKey,
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          sideshiftAccountId,
          amount: amountUsd * 0.20, // 20% Revenue Share to UGC Creator
          currency: 'USD',
          description: 'Soulpad UGC Creator Referral Commission',
        }),
      });
      return response.ok;
    } catch (error) {
      console.error('[SideShift Transfer Error]', error);
      return false;
    }
  }

export class MonetizationService {
  static async initialize(userId: string): Promise<void> {
    try {
      // 1. RevenueCat Configuration
      Purchases.setLogLevel(LOG_LEVEL.DEBUG);
      Purchases.configure({
        apiKey: 'goog_revenuecat_api_key', // Replace with iOS/Android keys
        appUserID: userId,
      });

      // 2. Superwall Configuration
      Superwall.configure('pk_superwall_public_key');

      console.log('[Soulpad] Monetization SDKs successfully bound.');
    } catch (error) {
      console.error('[Soulpad Monetization Init Error]', error);
    }
  }

  static async triggerPaywall(placementName: string): Promise<void> {
    try {
      // Direct integration with Superwall Paywall engine
      await Superwall.shared.register(placementName);
    } catch (error) {
      console.error(`[Superwall] Trigger error for placement ${placementName}:`, error);
    }
  }

  static async checkProStatus(): Promise<boolean> {
    try {
      const customerInfo = await Purchases.getCustomerInfo();
      return typeof customerInfo.entitlements.active['Soulpad Pass'] !== 'undefined';
    } catch (e) {
      return false;
    }
  }
}
